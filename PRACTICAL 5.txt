    <script>

        // Store selected food
        let selectedFood = "";
        let selectedPrice = 0;


        // Open Order Form
        function openOrder(food, price) {

            selectedFood = food;
            selectedPrice = price;

            // Display food details
            document.getElementById("foodName").innerText = food;

            document.getElementById("foodPrice").innerText = price;


            // Show order section
            document.getElementById("orderSection")
                .classList.remove("hidden");


            // Hide success message
            document.getElementById("successSection")
                .classList.add("hidden");


            // Scroll to form
            document.getElementById("orderSection")
                .scrollIntoView({
                    behavior: "smooth"
                });

        }


        // Place Order
        function placeOrder(event) {

            event.preventDefault();


            let name =
                document.getElementById("customerName").value;

            let quantity =
                document.getElementById("quantity").value;


            let total =
                selectedPrice * quantity;


            // Hide form
            document.getElementById("orderSection")
                .classList.add("hidden");


            // Show success
            document.getElementById("successSection")
                .classList.remove("hidden");


            // Show order details
            document.getElementById("orderDetails").innerText =
                "Thank you " + name +
                "! Your order for " +
                quantity + " × " +
                selectedFood +
                " has been placed. Total: ₹" +
                total;


            // Scroll to success
            document.getElementById("successSection")
                .scrollIntoView({
                    behavior: "smooth"
                });

        }


        // Cancel Order
        function closeOrder() {

            document.getElementById("orderSection")
                .classList.add("hidden");

        }


        // Order Again
        function newOrder() {

            document.getElementById("successSection")
                .classList.add("hidden");

            document.querySelector("section")
                .scrollIntoView({
                    behavior: "smooth"
                });

        }

    </script>
