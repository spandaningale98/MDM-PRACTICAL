    <script>

        // Get HTML elements
        const taskInput =
            document.getElementById("taskInput");

        const subjectInput =
            document.getElementById("subjectInput");

        const dateInput =
            document.getElementById("dateInput");

        const priorityInput =
            document.getElementById("priorityInput");

        const addBtn =
            document.getElementById("addBtn");

        const taskList =
            document.getElementById("taskList");

        const errorMessage =
            document.getElementById("errorMessage");

        const searchInput =
            document.getElementById("searchInput");

        const filterInput =
            document.getElementById("filterInput");

        const clearBtn =
            document.getElementById("clearBtn");

        const emptyMessage =
            document.getElementById("emptyMessage");


        // Add task
        addBtn.addEventListener("click", addTask);


        // Search task
        searchInput.addEventListener("input", filterTasks);


        // Filter task
        filterInput.addEventListener("change", filterTasks);


        // Add task function
        function addTask() {

            const task =
                taskInput.value.trim();

            const subject =
                subjectInput.value;

            const date =
                dateInput.value;

            const priority =
                priorityInput.value;


            // Validation
            if (task === "" ||
                subject === "" ||
                date === "") {

                errorMessage.textContent =
                    "Please fill all required fields.";

                return;
            }


            errorMessage.textContent = "";


            // Create task object
            const taskData = {
                task: task,
                subject: subject,
                date: date,
                priority: priority,
                completed: false
            };


            // Create task card
            createTask(taskData);


            // Clear inputs
            taskInput.value = "";
            subjectInput.value = "";
            dateInput.value = "";
            priorityInput.value = "Low";


            // Update statistics
            updateStats();

        }


        // Create task card
        function createTask(data) {

            const card =
                document.createElement("div");


            card.className =
                "task-card border rounded-xl p-4 " +
                "hover:shadow-md transition";


            // Top section
            const top =
                document.createElement("div");

            top.className =
                "flex flex-col md:flex-row " +
                "md:items-center md:justify-between gap-3";


            // Left section
            const left =
                document.createElement("div");

            left.className =
                "flex items-start gap-3";


            // Checkbox
            const checkbox =
                document.createElement("input");

            checkbox.type = "checkbox";

            checkbox.className =
                "w-5 h-5 mt-1 accent-indigo-600";


            // Task information
            const info =
                document.createElement("div");


            // Task title
            const title =
                document.createElement("h3");

            title.textContent =
                data.task;

            title.className =
                "font-bold text-lg task-title";


            // Subject
            const subject =
                document.createElement("p");

            subject.textContent =
                "📘 " + data.subject;

            subject.className =
                "text-sm text-gray-500";


            // Date
            const deadline =
                document.createElement("p");

            deadline.textContent =
                "📅 Deadline: " + data.date;

            deadline.className =
                "text-sm text-gray-500";


            info.appendChild(title);
            info.appendChild(subject);
            info.appendChild(deadline);


            left.appendChild(checkbox);
            left.appendChild(info);


            // Right section
            const right =
                document.createElement("div");

            right.className =
                "flex items-center gap-2";


            // Priority
            const priority =
                document.createElement("span");

            priority.textContent =
                data.priority;

            priority.className =
                "px-3 py-1 rounded-full text-sm font-semibold";


            if (data.priority === "High") {

                priority.classList.add(
                    "bg-red-100",
                    "text-red-600"
                );

            }
            else if (data.priority === "Medium") {

                priority.classList.add(
                    "bg-yellow-100",
                    "text-yellow-600"
                );

            }
            else {

                priority.classList.add(
                    "bg-green-100",
                    "text-green-600"
                );

            }


            // Delete button
            const deleteBtn =
                document.createElement("button");

            deleteBtn.textContent =
                "Delete";

            deleteBtn.className =
                "bg-red-500 hover:bg-red-600 " +
                "text-white px-3 py-2 rounded-lg text-sm";


            right.appendChild(priority);
            right.appendChild(deleteBtn);


            top.appendChild(left);
            top.appendChild(right);


            card.appendChild(top);


            // Complete task
            checkbox.addEventListener(
                "change",
                function() {

                    title.classList.toggle(
                        "line-through",
                        checkbox.checked
                    );

                    title.classList.toggle(
                        "text-gray-400",
                        checkbox.checked
                    );

                    updateStats();

                    filterTasks();
                }
            );


            // Delete task
            deleteBtn.addEventListener(
                "click",
                function() {

                    card.remove();

                    updateStats();

                    filterTasks();

                }
            );


            taskList.appendChild(card);

            emptyMessage.style.display = "none";

        }


        // Update statistics
        function updateStats() {

            const cards =
                taskList.querySelectorAll(".task-card");


            let completed = 0;


            cards.forEach(function(card) {

                const checkbox =
                    card.querySelector("input[type='checkbox']");


                if (checkbox.checked) {
                    completed++;
                }

            });


            const total = cards.length;

            const pending = total - completed;


            document.getElementById(
                "totalTasks"
            ).textContent = total;


            document.getElementById(
                "completedTasks"
            ).textContent = completed;


            document.getElementById(
                "pendingTasks"
            ).textContent = pending;


            if (total === 0) {

                emptyMessage.style.display = "block";

            }
            else {

                emptyMessage.style.display = "none";

            }

        }


        // Search and filter
        function filterTasks() {

            const search =
                searchInput.value.toLowerCase();

            const filter =
                filterInput.value;


            const cards =
                taskList.querySelectorAll(".task-card");


            let visible = 0;


            cards.forEach(function(card) {

                const title =
                    card.querySelector(
                        ".task-title"
                    ).textContent.toLowerCase();


                const checkbox =
                    card.querySelector(
                        "input[type='checkbox']"
                    );


                const matchesSearch =
                    title.includes(search);


                let matchesFilter = true;


                if (filter === "Completed") {

                    matchesFilter =
                        checkbox.checked;

                }


                if (filter === "Pending") {

                    matchesFilter =
                        !checkbox.checked;

                }


                if (matchesSearch &&
                    matchesFilter) {

                    card.style.display = "block";

                    visible++;

                }
                else {

                    card.style.display = "none";

                }

            });


            if (visible === 0) {

                emptyMessage.style.display = "block";

            }
            else {

                emptyMessage.style.display = "none";

            }

        }


        // Clear all tasks
        clearBtn.addEventListener(
            "click",
            function() {

                if (taskList.children.length === 0) {
                    return;
                }


                if (confirm(
                    "Delete all study tasks?"
                )) {

                    taskList.innerHTML = "";

                    updateStats();

                }

            }
        );
    </script>
