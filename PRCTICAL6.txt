    <script>

        function calculateGrade() {

            // Get marks
            let math = Number(document.getElementById("math").value);
            let java = Number(document.getElementById("java").value);
            let dsa = Number(document.getElementById("dsa").value);
            let python = Number(document.getElementById("python").value);
            let web = Number(document.getElementById("web").value);


            // Validate marks
            if (
                math < 0 || math > 100 ||
                java < 0 || java > 100 ||
                dsa < 0 || dsa > 100 ||
                python < 0 || python > 100 ||
                web < 0 || web > 100
            ) {

                alert("Please enter marks between 0 and 100.");
                return;

            }


            // Calculate total
            let total = math + java + dsa + python + web;


            // Calculate percentage
            let percentage = total / 5;


            // Calculate grade
            let grade;

            if (percentage >= 90) {
                grade = "A+";
            }
            else if (percentage >= 80) {
                grade = "A";
            }
            else if (percentage >= 70) {
                grade = "B";
            }
            else if (percentage >= 60) {
                grade = "C";
            }
            else if (percentage >= 50) {
                grade = "D";
            }
            else {
                grade = "F";
            }


            // Pass / Fail
            let status;

            if (percentage >= 40) {
                status = "PASS";
            }
            else {
                status = "FAIL";
            }


            // Display result
            document.getElementById("total").innerText = total;

            document.getElementById("percentage").innerText =
                percentage.toFixed(2);

            document.getElementById("grade").innerText = grade;

            document.getElementById("status").innerText = status;


            // Show result box
            document.getElementById("result")
                .classList.remove("hidden");

        }

    </script>
